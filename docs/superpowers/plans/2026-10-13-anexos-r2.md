# Anexos em todas as conversas + storage no R2 — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** anexo funcionando nos quatro caminhos (pessoa↔pessoa, pessoa→agente no chat e no canal, agente→pessoa), privado, até 100 MB, guardado no Cloudflare R2 — e o disco da VPS (`UPLOADS_DIR`) deixando de ser usado.

**Architecture:** um módulo único (`app/armazenamento.py`) fala com o R2 pela API S3; o backend é a porta (confere permissão, emite URL assinada) e nunca o cano (bytes vão do navegador direto ao R2 e de volta). Anexo de conversa vira linha na tabela `anexos`, que decide quem lê. O agente lê anexo por um servidor MCP (`hsos-anexos`) cuja URL carrega um HMAC do `agentId` — a identidade do agente é verificada no servidor, não declarada por ele.

**Tech Stack:** FastAPI + asyncpg, `boto3` (S3 → R2), `openpyxl` (já instalado), `xlrd` (novo, `.xls` legado), PyJWT (já instalado, para o ticket de upload); React + Vite + Vitest.

# backend/app/armazenamento.py
"""Onde os arquivos moram — Cloudflare R2, pela API S3.

Único módulo que conhece o R2. Rotas, relatórios e o servidor MCP de anexos
falam com ele por `chave(bucket, caminho)` e pelas funções `async` abaixo;
nenhum outro lugar importa `boto3`.

Um bucket físico só (`R2_BUCKET`), com o bucket **lógico** como primeiro
segmento da chave. A permissão é do bucket lógico e mora nas rotas; aqui só se
guarda e se assina.

⚠️ **O `boto3` é síncrono.** Toda chamada roda em thread (`anyio.to_thread`),
senão um upload lento do R2 trava o event loop e, com ele, todo request
daquele worker — chat incluído.
"""
from __future__ import annotations

import logging
import re
from dataclasses import dataclass
from urllib.parse import quote

import anyio
from fastapi import HTTPException, status

from app.config import settings

logger = logging.getLogger(__name__)

BUCKETS_PUBLICOS = {"agent-files", "audio-messages", "wiki-uploads"}
BUCKETS_PRIVADOS = {"company-docs", "generated-documents", "anexos"}
BUCKETS = BUCKETS_PUBLICOS | BUCKETS_PRIVADOS

TAMANHO_MAXIMO = 100 * 1024 * 1024

# Segmento aceitável: letras, números, ponto, hífen, sublinhado. O regex aceita
# `.` e `..` — no disco quem barrava a travessia era o `resolve()`; na chave é
# a conferência explícita em `chave()`.
_SEGMENTO = re.compile(r"^[A-Za-z0-9._-]{1,120}$")


def chave(bucket: str, caminho: str) -> str:
    if bucket not in BUCKETS:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "Bucket desconhecido.")
    partes = [p for p in (caminho or "").split("/") if p]
    if not partes or any(p in (".", "..") or not _SEGMENTO.match(p) for p in partes):
        raise HTTPException(status.HTTP_400_BAD_REQUEST, "Caminho de arquivo inválido.")
    return f"{bucket}/{'/'.join(partes)}"


@dataclass(frozen=True)
class Objeto:
    tamanho: int
    mime: str | None


class R2:
    def __init__(self, account_id: str, chave_id: str, segredo: str, bucket: str):
        import boto3
        from botocore.config import Config

        self.bucket = bucket
        self.s3 = boto3.client(
            "s3",
            endpoint_url=f"https://{account_id}.r2.cloudflarestorage.com",
            aws_access_key_id=chave_id,
            aws_secret_access_key=segredo,
            region_name="auto",
            config=Config(signature_version="s3v4", s3={"addressing_style": "path"}),
        )

    @staticmethod
    def _nao_existe(e: Exception) -> bool:
        codigo = getattr(e, "response", {}).get("Error", {}).get("Code", "")
        return codigo in ("404", "NoSuchKey", "NotFound")

    def gravar(self, chave: str, dados: bytes, mime: str) -> None:
        self.s3.put_object(Bucket=self.bucket, Key=chave, Body=dados, ContentType=mime)

    def ler(self, chave: str) -> bytes | None:
        try:
            return self.s3.get_object(Bucket=self.bucket, Key=chave)["Body"].read()
        except Exception as e:  # noqa: BLE001
            if self._nao_existe(e):
                return None
            raise

    def cabecalho(self, chave: str) -> Objeto | None:
        try:
            r = self.s3.head_object(Bucket=self.bucket, Key=chave)
        except Exception as e:  # noqa: BLE001
            if self._nao_existe(e):
                return None
            raise
        return Objeto(tamanho=int(r["ContentLength"]), mime=r.get("ContentType"))

    def url_get(self, chave: str, segundos: int, nome_download: str | None, inline: bool) -> str:
        params = {"Bucket": self.bucket, "Key": chave}
        if nome_download:
            modo = "inline" if inline else "attachment"
            params["ResponseContentDisposition"] = f"{modo}; filename*=UTF-8''{quote(nome_download)}"
        return self.s3.generate_presigned_url("get_object", Params=params, ExpiresIn=segundos)

    def url_put(self, chave: str, mime: str, tamanho: int, segundos: int) -> str:
        # `ContentLength` entra na assinatura: o R2 recusa corpo de tamanho
        # diferente do declarado. O `concluir` confere de novo do lado de cá.
        return self.s3.generate_presigned_url(
            "put_object",
            Params={"Bucket": self.bucket, "Key": chave, "ContentType": mime,
                    "ContentLength": tamanho},
            ExpiresIn=segundos,
        )

    def apagar(self, chave: str) -> None:
        self.s3.delete_object(Bucket=self.bucket, Key=chave)

    def listar(self, prefixo: str, recursivo: bool) -> list[dict]:
        pag = self.s3.get_paginator("list_objects_v2")
        extra = {} if recursivo else {"Delimiter": "/"}
        itens = []
        for pagina in pag.paginate(Bucket=self.bucket, Prefix=prefixo, **extra):
            for o in pagina.get("Contents", []):
                itens.append({"chave": o["Key"], "nome": o["Key"].rsplit("/", 1)[-1],
                              "tamanho": int(o["Size"])})
        return itens


class Memoria:
    """Backend em memória com a mesma interface — para teste, nunca produção."""

    def __init__(self):
        self.objetos: dict[str, tuple[bytes, str]] = {}

    def gravar(self, chave, dados, mime):
        self.objetos[chave] = (dados, mime)

    def ler(self, chave):
        o = self.objetos.get(chave)
        return o[0] if o else None

    def cabecalho(self, chave):
        o = self.objetos.get(chave)
        return Objeto(tamanho=len(o[0]), mime=o[1]) if o else None

    def url_get(self, chave, segundos, nome_download, inline):
        return f"https://r2.invalido/{chave}?expira={segundos}"

    def url_put(self, chave, mime, tamanho, segundos):
        return f"https://r2.invalido/{chave}?put=1&tamanho={tamanho}&expira={segundos}"

    def apagar(self, chave):
        self.objetos.pop(chave, None)

    def listar(self, prefixo, recursivo):
        itens = []
        for k, (dados, _) in sorted(self.objetos.items()):
            if not k.startswith(prefixo):
                continue
            if not recursivo and "/" in k[len(prefixo):]:
                continue
            itens.append({"chave": k, "nome": k.rsplit("/", 1)[-1], "tamanho": len(dados)})
        return itens


_backend = None


def definir(backend) -> None:
    global _backend
    _backend = backend


def _obter():
    global _backend
    if _backend is not None:
        return _backend
    faltando = [n for n in ("R2_ACCOUNT_ID", "R2_ACCESS_KEY_ID", "R2_SECRET_ACCESS_KEY", "R2_BUCKET")
                if not getattr(settings, n)]
    if faltando:
        raise HTTPException(
            status.HTTP_503_SERVICE_UNAVAILABLE,
            f"Armazenamento de arquivos não configurado: falta {', '.join(faltando)}.",
        )
    _backend = R2(settings.R2_ACCOUNT_ID, settings.R2_ACCESS_KEY_ID,
                  settings.R2_SECRET_ACCESS_KEY, settings.R2_BUCKET)
    return _backend


async def _rodar(metodo: str, *args):
    backend = _obter()
    try:
        return await anyio.to_thread.run_sync(getattr(backend, metodo), *args)
    except HTTPException:
        raise
    except Exception as e:  # noqa: BLE001
        logger.error("R2 falhou em %s(%s): %s", metodo, args[0] if args else "", e)
        raise HTTPException(
            status.HTTP_502_BAD_GATEWAY,
            "O armazenamento de arquivos (R2) recusou a operação. Tente de novo; "
            "se persistir, avise o administrador.",
        )


async def gravar(chave: str, dados: bytes, mime: str) -> None:
    await _rodar("gravar", chave, dados, mime)


async def ler(chave: str) -> bytes | None:
    return await _rodar("ler", chave)


async def cabecalho(chave: str) -> Objeto | None:
    return await _rodar("cabecalho", chave)


async def url_get(chave: str, segundos: int, nome_download: str | None = None,
                  inline: bool = True) -> str:
    return await _rodar("url_get", chave, segundos, nome_download, inline)


async def url_put(chave: str, mime: str, tamanho: int, segundos: int) -> str:
    return await _rodar("url_put", chave, mime, tamanho, segundos)


async def apagar(chave: str) -> None:
    await _rodar("apagar", chave)


async def listar(prefixo: str, recursivo: bool = False) -> list[dict]:
    return await _rodar("listar", prefixo, recursivo)
```

/**
 * Sobe direto para o R2 com o PUT assinado, reportando progresso.
 *
 * `XMLHttpRequest` e não `fetch` porque `fetch` não tem progresso de envio —
 * e com 100 MB a pessoa precisa ver que está andando.
 *
 * ⚠️ O `Content-Type` tem que ser EXATAMENTE o que foi assinado no `iniciar`.
 * Outro valor e o R2 responde 403 SignatureDoesNotMatch.
 */
function putDiretoImpl(url: string, arquivo: Blob, mime: string, aoProgredir?: (f: number) => void): Promise<void> {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open("PUT", url);
    xhr.setRequestHeader("Content-Type", mime);
    xhr.upload.onprogress = (e) => {
      if (e.lengthComputable && aoProgredir) aoProgredir(e.loaded / e.total);
    };
    xhr.onload = () =>
      xhr.status >= 200 && xhr.status < 300
        ? resolve()
        : reject(new Error(`O armazenamento recusou o envio (HTTP ${xhr.status}).`));
    // Erro de rede aqui é quase sempre CORS do bucket — o navegador não deixa
    // ver o motivo. Dizer isso poupa a tarde de procurar no backend.
    xhr.onerror = () => reject(new Error("Falha ao enviar o arquivo ao armazenamento (rede ou CORS do bucket)."));
    xhr.send(arquivo);
  });
}

export const interno = { putDireto: putDiretoImpl };

export function putDireto(url: string, arquivo: Blob, mime: string, aoProgredir?: (f: number) => void) {
  return interno.putDireto(url, arquivo, mime, aoProgredir);
}

    for arquivo in sorted(ORIGEM.rglob("*")):
        if not arquivo.is_file():
            continue
        rel = arquivo.relative_to(ORIGEM).as_posix()
        bucket = rel.split("/", 1)[0]
        if bucket not in BUCKETS:
            print(f"  ⚠️  fora de bucket conhecido, ignorado: {rel}")
            continue
        tamanho = arquivo.stat().st_size
        disco[bucket][0] += 1
        disco[bucket][1] += tamanho
        ja = r2.cabecalho(rel)
        if ja is not None and ja.tamanho == tamanho:
            pulados += 1
            continue
        if aplicar:
            mime = mimetypes.guess_type(arquivo.name)[0] or "application/octet-stream"
            r2.gravar(rel, arquivo.read_bytes(), mime)
        enviados += 1

    print(f"\n{'enviados' if aplicar else 'a enviar'}: {enviados} · já no R2: {pulados}\n")
    print(f"  {'bucket':22} {'disco':>16} {'R2':>16}")
    ok = True
    for bucket in sorted(BUCKETS):
        n_d, b_d = disco.get(bucket, [0, 0])
        itens = r2.listar(f"{bucket}/", recursivo=True)
        n_r, b_r = len(itens), sum(i["tamanho"] for i in itens)
        marca = "✅" if (n_d, b_d) == (n_r, b_r) or n_r >= n_d and b_r >= b_d else "❌"
        ok &= marca == "✅"
        print(f"  {bucket:22} {n_d:>6} · {b_d:>8} {n_r:>6} · {b_r:>8}  {marca}")
    if not aplicar:
        print("\n  (ensaio — nada foi escrito; rode com --aplicar)")
        return 0
    print("\n  " + ("bateu — pode tirar o volume do EasyPanel" if ok else "NÃO bateu — não tire o volume"))
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main())
```

# backend/app/anexos.py
"""Anexo de conversa: quem lê, onde mora, como o agente o vê.

Funções puras — a decisão de acesso fica aqui, testada sem banco, e as rotas
(`app/routers/anexos.py`, `app/routers/mcp_anexos.py`) só buscam os fatos
(a linha, se é membro do canal) e perguntam.
"""
from __future__ import annotations

import re
import unicodedata
from dataclasses import dataclass

from app.gateway.client import normalizar_agent_id

SQL_ANEXO = """
    SELECT id::text AS id, caminho, nome, mime, tamanho, contexto,
           channel_id::text AS channel_id, agent_id, dono_conversa::text AS dono_conversa
      FROM public.anexos WHERE id = $1::uuid
"""


@dataclass(frozen=True)
class Anexo:
    id: str
    caminho: str
    nome: str
    mime: str
    tamanho: int
    contexto: str
    channel_id: str | None
    agent_id: str | None
    dono_conversa: str | None

    @classmethod
    def de_linha(cls, linha) -> "Anexo":
        return cls(**{k: linha[k] for k in cls.__dataclass_fields__})


def pessoa_pode_ler(anexo: Anexo, usuario_id: str, membro_do_canal: bool) -> bool:
    if anexo.contexto == "canal":
        return membro_do_canal
    if anexo.contexto == "conversa_agente":
        return bool(anexo.dono_conversa) and anexo.dono_conversa == usuario_id
    return False


def agente_pode_ler(anexo: Anexo, agent_id: str, agente_no_canal: bool) -> bool:
    if anexo.contexto == "canal":
        return agente_no_canal
    if anexo.contexto == "conversa_agente":
        return bool(anexo.agent_id) and \
            normalizar_agent_id(anexo.agent_id) == normalizar_agent_id(agent_id)
    return False


def nome_saneado(nome: str) -> str:
    """Nome que passa no `_SEGMENTO` do armazenamento, legível o bastante.

    O nome original continua na tabela e é o que aparece na tela e no
    download — este só serve de chave.
    """
    sem_acento = unicodedata.normalize("NFKD", nome or "").encode("ascii", "ignore").decode()
    base, ponto, ext = sem_acento.rpartition(".")
    if not ponto:
        base, ext = sem_acento, ""
    limpo = lambda s: re.sub(r"-{2,}", "-", re.sub(r"[^A-Za-z0-9._-]", "-", s)).strip("-.")  # noqa: E731
    base, ext = limpo(base), limpo(ext)[:10]
    if not base:
        return "arquivo"
    sufixo = f".{ext}" if ext else ""
    return base[: 100 - len(sufixo)] + sufixo

/**
 * Anexo de conversa — privado, lido por quem participa da conversa.
 *
 * Desde 13/10/2026 anexo não vai mais para o bucket público `agent-files`:
 * vai para `anexos`, e a URL para exibir é pedida ao backend, que confere se
 * a pessoa participa. Como `<img src>` não manda token, a tela pede a URL
 * assinada antes e a guarda até perto de vencer.
 *
 * Anexo antigo (sem `id`) continua com a `url` pública de sempre — nenhuma
 * URL gravada foi reescrita.
 */
import { api } from "@/lib/api";
import { putDireto } from "@/lib/storage";

export type DestinoAnexo =
  | { contexto: "canal"; channelId: string }
  | { contexto: "conversa_agente"; agentId: string };

export interface AnexoEnviado {
  id: string;
  name: string;
  size: number;
  mimeType: string;
  url: string;
}

export async function enviarAnexo(
  destino: DestinoAnexo,
  arquivo: File,
  aoProgredir?: (fracao: number) => void,
): Promise<AnexoEnviado> {
  const mime = arquivo.type || "application/octet-stream";
  const inicio = await api<{ url_put: string; ticket: string }>("/storage/upload/iniciar", {
    method: "POST",
    body: {
      bucket: "anexos",
      nome: arquivo.name,
      mime,
      tamanho: arquivo.size,
      contexto: destino.contexto,
      ...(destino.contexto === "canal" ? { channel_id: destino.channelId } : { agent_id: destino.agentId }),
    },
  });
  await putDireto(inicio.url_put, arquivo, mime, aoProgredir);
  return api<AnexoEnviado>("/storage/upload/concluir", { method: "POST", body: { ticket: inicio.ticket } });
}

const MARGEM_MS = 30_000;
const cache = new Map<string, { url: string; venceEm: number }>();
const emVoo = new Map<string, Promise<string>>();

export async function urlDoAnexo(id: string, agora: () => number = Date.now): Promise<string> {
  const guardada = cache.get(id);
  if (guardada && guardada.venceEm - agora() > MARGEM_MS) return guardada.url;
  const pendente = emVoo.get(id);
  if (pendente) return pendente;
  const p = api<{ url: string; expira_em: number }>(`/anexos/${encodeURIComponent(id)}/url`)
    .then((r) => {
      cache.set(id, { url: r.url, venceEm: agora() + r.expira_em * 1000 });
      return r.url;
    })
    .finally(() => emVoo.delete(id));
  emVoo.set(id, p);
  return p;
}

export function esquecerUrl(id: string): void {
  cache.delete(id);
}

interface Props {
  id?: string;
  url?: string;
  children: (url: string, renovar: () => void) => ReactNode;
}

export default function AnexoResolvido({ id, url, children }: Props) {
  const [resolvida, setResolvida] = useState<string | null>(id ? null : url ?? "");
  const [erro, setErro] = useState(false);
  const [tentativa, setTentativa] = useState(0);

  useEffect(() => {
    if (!id) return;
    let vivo = true;
    urlDoAnexo(id)
      .then((u) => vivo && setResolvida(u))
      .catch(() => vivo && setErro(true));
    return () => {
      vivo = false;
    };

    if (!id || tentativa > 0) return;
    esquecerUrl(id);
    setTentativa(1);

  if (resolvida === null) return <div className="h-9 w-40 animate-pulse rounded-lg bg-secondary/50" />;
  return <>{children(resolvida, renovar)}</>;

```python
# backend/app/anexos_leitura.py
"""Transforma um anexo em texto que o agente consegue ler, por páginas.

Saída em TSV (tabulação entre colunas): é o formato que menos gasta token e
que o modelo lê sem confundir vírgula decimal brasileira com separador.
"""
from __future__ import annotations

import csv
import io
from datetime import date, datetime, time

from app.anexos import tipo_legivel

LINHAS_POR_PAGINA = 200
CARACTERES_POR_PAGINA = 20_000


def _celula(v) -> str:
    if v is None:
        return ""
    if isinstance(v, datetime):
        return v.strftime("%Y-%m-%d %H:%M") if (v.hour or v.minute) else v.strftime("%Y-%m-%d")
    if isinstance(v, (date, time)):
        return v.isoformat()
    if isinstance(v, float) and v.is_integer():
        return str(int(v))
    return str(v).replace("\t", " ").replace("\n", " ").strip()


def _abas_xlsx(dados: bytes) -> dict[str, list[list[str]]]:
    import openpyxl

    # `data_only=True`: o valor calculado, não a fórmula. Exportação de
    # sistema (Flash) já vem com valor; fórmula crua não diz nada ao agente.
    wb = openpyxl.load_workbook(io.BytesIO(dados), read_only=True, data_only=True)
    return {ws.title: [[_celula(c) for c in linha] for linha in ws.iter_rows(values_only=True)]
            for ws in wb.worksheets}


def _abas_xls(dados: bytes) -> dict[str, list[list[str]]]:
    import xlrd

    wb = xlrd.open_workbook(file_contents=dados)
    abas = {}
    for sh in wb.sheets():
        linhas = []
        for r in range(sh.nrows):
            linha = []
            for c in range(sh.ncols):
                cel = sh.cell(r, c)
                if cel.ctype == xlrd.XL_CELL_DATE:
                    linha.append(_celula(xlrd.xldate_as_datetime(cel.value, wb.datemode)))
                else:
                    linha.append(_celula(cel.value))
            linhas.append(linha)
        abas[sh.name] = linhas
    return abas


def _decodificar(dados: bytes) -> str:
    # Ordem importa: UTF-8 estrito falha em cp1252 com acento; cp1252 nunca falha.
    for cod in ("utf-8-sig", "cp1252"):
        try:
            return dados.decode(cod)
        except UnicodeDecodeError:
            continue
    return dados.decode("latin-1")


def _abas_csv(dados: bytes) -> dict[str, list[list[str]]]:
    texto = _decodificar(dados)
    amostra = texto[:5000]
    try:
        dialeto = csv.Sniffer().sniff(amostra, delimiters=";,\t|")
    except csv.Error:
        dialeto = csv.excel
        dialeto.delimiter = ";" if amostra.count(";") > amostra.count(",") else ","
    return {"CSV": [[_celula(c) for c in l] for l in csv.reader(io.StringIO(texto), dialeto)]}


def _sem_linhas_vazias(linhas: list[list[str]]) -> list[list[str]]:
    return [l for l in linhas if any(c for c in l)]


def _planilha(nome: str, dados: bytes, aba: str | None, pagina: int) -> str:
    ext = nome.rsplit(".", 1)[-1].lower()
    leitor = {"xlsx": _abas_xlsx, "xlsm": _abas_xlsx, "xls": _abas_xls, "csv": _abas_csv}.get(ext)
    if leitor is None:
        return f"Formato de planilha .{ext} não suportado. Peça o arquivo em .xlsx, .xls ou .csv."
    abas = {n: _sem_linhas_vazias(l) for n, l in leitor(dados).items()}
    resumo = "Abas: " + ", ".join(f"{n} ({len(l)} linhas)" for n, l in abas.items())
    if aba is not None and aba not in abas:
        return f"A aba \"{aba}\" não existe. {resumo}"
    nome_aba = aba or next(iter(abas), None)
    if nome_aba is None:
        return "Planilha sem nenhuma aba com conteúdo."
    linhas = abas[nome_aba]
    if not linhas:
        return f"{resumo}\nA aba {nome_aba} está vazia."
    cabecalho, corpo = linhas[0], linhas[1:]
    total = max(1, -(-len(corpo) // LINHAS_POR_PAGINA))
    if pagina > total:
        return f"{resumo}\nA aba {nome_aba} só tem {total} página{'s' if total > 1 else ''}."
    ini = (pagina - 1) * LINHAS_POR_PAGINA
    fatia = corpo[ini: ini + LINHAS_POR_PAGINA]
    return "\n".join([
        resumo,
        f"Aba: {nome_aba} · página {pagina}/{total} · linhas {ini + 1}–{ini + len(fatia)} de {len(corpo)}"
        " (a 1ª linha da planilha é o cabeçalho e se repete em toda página)",
        "\t".join(cabecalho),
        *("\t".join(l) for l in fatia),
    ])


def _texto(nome: str, dados: bytes) -> str:
    ext = nome.rsplit(".", 1)[-1].lower()
    if ext == "pdf":
        from pypdf import PdfReader

        return "\n".join((p.extract_text() or "") for p in PdfReader(io.BytesIO(dados)).pages)
    if ext == "docx":
        import docx

        return "\n".join(p.text for p in docx.Document(io.BytesIO(dados)).paragraphs)
    return _decodificar(dados)


def _paginar_texto(texto: str, pagina: int) -> str:
    total = max(1, -(-len(texto) // CARACTERES_POR_PAGINA))
    if pagina > total:
        return f"O texto só tem {total} página{'s' if total > 1 else ''}."
    ini = (pagina - 1) * CARACTERES_POR_PAGINA
    return f"Texto · página {pagina}/{total}\n\n" + texto[ini: ini + CARACTERES_POR_PAGINA]


def conteudo_legivel(nome: str, mime: str, dados: bytes, aba: str | None = None, pagina: int = 1) -> str:
    pagina = max(1, int(pagina or 1))
    tipo = tipo_legivel(nome, mime)
    if tipo == "planilha":
        return _planilha(nome, dados, aba, pagina)
    if tipo in ("pdf", "documento", "texto"):
        texto = _texto(nome, dados)
        if not texto.strip():
            return "Arquivo sem texto extraível (PDF escaneado precisa de OCR)."
        return _paginar_texto(texto, pagina)
    return (f"Arquivo {nome} ({tipo}) — o conteúdo não é legível por ferramenta. "
            "Diga à pessoa o que você consegue fazer com ele e peça outro formato se precisar.")
```

⚠️ **A identidade vem da URL, não do argumento.** Cada agente tem no gateway
uma entrada própria (`hsos-anexos-<id>`) cuja URL termina em

bateu, 404 — sem confirmar que a rota existe. É o que separa este servidor
dos outros três, que confiam no `agente` que o próprio agente declara.

import logging

from app import anexos as ax
from app import armazenamento as arm

def assinatura_do_agente(segredo: str, agent_id: str) -> str:
    return hmac.new(segredo.encode(), normalizar_agent_id(agent_id).encode(), hashlib.sha256).hexdigest()


def assinatura_confere(segredo: str, agent_id: str, assinatura: str) -> bool:
    if not segredo:
        return False
    return hmac.compare_digest(assinatura_do_agente(segredo, agent_id), assinatura or "")

def _resposta(ident, resultado=None, erro=None):
    corpo = {"jsonrpc": "2.0", "id": ident}
    corpo["error" if erro else "result"] = erro or resultado
    return JSONResponse(corpo)


def _texto(ident, texto: str, erro: bool = False):
    return _resposta(ident, {"content": [{"type": "text", "text": texto}], "isError": erro})


async def _ler_anexo(agent_id: str, args: dict) -> tuple[str, bool]:
    anexo_id = str(args.get("id") or "").strip()
    async with sessao(role="service_role") as conn:
        anexo = await buscar_anexo(conn, anexo_id)
        no_canal = bool(anexo and anexo.contexto == "canal"
                        and await membro_do_canal(conn, anexo.channel_id, agent_id, "agent"))
    if anexo is None or not ax.agente_pode_ler(anexo, agent_id, no_canal):
        logger.warning("ler_anexo recusado: agente %s, anexo %s", agent_id, anexo_id)
        return ("Este anexo não pertence a uma conversa sua. Não tente de novo; "
                "peça à pessoa para anexar o arquivo aqui."), True
    dados = await arm.ler(arm.chave("anexos", anexo.caminho))
    if dados is None:
        return "O arquivo deste anexo não está mais no armazenamento.", True
    try:
        texto = conteudo_legivel(anexo.nome, anexo.mime, dados,
                                 aba=args.get("aba"), pagina=int(args.get("pagina") or 1))
    except Exception as e:  # noqa: BLE001
        logger.warning("ler_anexo: %s não pôde ser lido: %s", anexo.nome, e)
        return f"Não consegui ler {anexo.nome}: o arquivo parece corrompido ou protegido por senha.", True
    logger.info("ler_anexo: %s leu %s (%s)", agent_id, anexo.id, anexo.nome)
    return f"{anexo.nome}\n{texto}", False


@router.post("/anexos/{agent_id}/{assinatura}")
async def mcp_anexos(agent_id: str, assinatura: str, request: Request,
                     _: None = Depends(exige_segredo("GUARDRAILS_API_TOKEN"))):
    segredo = await ler_segredo("ANEXOS_MCP_SEGREDO") or ""
    if not assinatura_confere(segredo, agent_id, assinatura):
        raise HTTPException(status.HTTP_404_NOT_FOUND, "Not Found")
    agent_id = normalizar_agent_id(agent_id)

    try:
        corpo = await request.json()
    except Exception:
        return _resposta(None, erro={"code": -32700, "message": "JSON inválido."})
    metodo, ident, params = corpo.get("method"), corpo.get("id"), corpo.get("params") or {}

    if ident is None:
        return JSONResponse({}, status_code=status.HTTP_202_ACCEPTED)
    if metodo == "initialize":
        return _resposta(ident, {"protocolVersion": _PROTOCOLO, "capabilities": {"tools": {}},
                                 "serverInfo": {"name": "hsos-anexos", "version": "1.0.0"}})
    if metodo == "tools/list":
        return _resposta(ident, {"tools": _FERRAMENTAS})
    if metodo == "tools/call":
        nome, args = params.get("name"), params.get("arguments") or {}
        if nome == "ler_anexo":
            texto, erro = await _ler_anexo(agent_id, args)
            return _texto(ident, texto, erro)
        return _resposta(ident, erro={"code": -32601, "message": f"Ferramenta desconhecida: {nome}"})
    return _resposta(ident, erro={"code": -32601, "message": f"Método não suportado: {metodo}"})

"""Monta o arquivo que o agente devolve, a partir de texto."""
from __future__ import annotations

import csv
import io
import re

from app.routers.storage import DefinicaoIn, SecaoIn, _montar_docx, _montar_pdf

FORMATOS = {"csv", "xlsx", "md", "docx", "pdf"}
LIMITE_CONTEUDO = 2_000_000

_MIME = {
    "csv": "text/csv",
    "xlsx": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
    "md": "text/markdown",
    "docx": "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    "pdf": "application/pdf",
}


def _linhas_csv(conteudo: str) -> list[list[str]]:
    amostra = conteudo[:5000]
    delim = ";" if amostra.count(";") >= amostra.count(",") else ","
    return [l for l in csv.reader(io.StringIO(conteudo), delimiter=delim) if any(l)]


def secoes_de_markdown(md: str) -> list[SecaoIn]:
    secoes = []
    for linha in md.splitlines():
        t = linha.strip()
        if not t:
            continue
        m = re.match(r"^(#{1,3})\s+(.*)$", t)
        if m:
            secoes.append(SecaoIn(heading=f"H{len(m.group(1))}", text=m.group(2).strip()))
            continue
        negrito = re.fullmatch(r"\*\*(.+)\*\*", t)
        secoes.append(SecaoIn(text=negrito.group(1) if negrito else t, bold=bool(negrito)))
    return secoes


def montar_arquivo(formato: str, nome: str, conteudo: str) -> tuple[bytes, str, str]:
    formato = (formato or "").lower().strip()
    if formato not in FORMATOS:
        raise ValueError(f"Formato {formato!r} não suportado. Use um de: {', '.join(sorted(FORMATOS))}.")
    if not (conteudo or "").strip():
        raise ValueError("Conteúdo vazio — nada para enviar.")
    if len(conteudo) > LIMITE_CONTEUDO:
        raise ValueError("Conteúdo grande demais para um arquivo enviado pelo chat (limite de 2 milhões de caracteres).")

    base = (nome or "arquivo").strip()
    if base.lower().endswith(f".{formato}"):
        base = base[: -len(formato) - 1]
    nome_final = f"{base or 'arquivo'}.{formato}"

    if formato == "csv":
        return conteudo.encode("utf-8-sig"), _MIME["csv"], nome_final
    if formato == "md":
        return conteudo.encode("utf-8"), _MIME["md"], nome_final
    if formato == "xlsx":
        import openpyxl

        wb = openpyxl.Workbook()
        ws = wb.active
        for l in _linhas_csv(conteudo):
            ws.append(l)
        b = io.BytesIO()
        wb.save(b)
        return b.getvalue(), _MIME["xlsx"], nome_final
    secoes = secoes_de_markdown(conteudo)
    titulo = next((s.text for s in secoes if s.heading == "H1"), base)
    definicao = DefinicaoIn(title=titulo, sections=[s for s in secoes if not (s.heading == "H1" and s.text == titulo)])
    dados = _montar_pdf(definicao) if formato == "pdf" else _montar_docx(definicao)
    return dados, _MIME[formato], nome_final
